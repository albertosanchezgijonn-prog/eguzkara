# Copia este archivo a  .github/workflows/build-apk.yml  en tu repositorio de GitHub
name: Build Android APK
on:
  workflow_dispatch:
  push:
    branches: [main]
jobs:
  apk:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: 20 }
      - uses: actions/setup-java@v4
        with: { distribution: temurin, java-version: 21 }
      - run: npm install
      - run: npm test
      - run: npm run web
      - run: npx cap add android
      - name: Permiso de micrófono
        run: |
          grep -q RECORD_AUDIO android/app/src/main/AndroidManifest.xml || \
          sed -i 's#</manifest>#    <uses-permission android:name="android.permission.RECORD_AUDIO" />\n</manifest>#' android/app/src/main/AndroidManifest.xml
      - run: npx cap sync android
      - run: cd android && chmod +x gradlew && ./gradlew assembleDebug
      - uses: actions/upload-artifact@v4
        with:
          name: eguzkara-debug-apk
          path: android/app/build/outputs/apk/debug/app-debug.apk
